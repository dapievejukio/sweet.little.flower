/* ===================================================== SWEET.LITTLE.FLOWER — MENSAGENS ===================================================== */ let inicioNoSite = null; let contadorBuscas = 0; const timersMensagens = []; const mensagensUsadas = new Set(); /* MENSAGENS PROGRAMADAS tempo em milissegundos: 60000 = 1 minuto 120000 = 2 minutos etc. */ const mensagensPorTempo = [ { tempo: 60000, texto: "vai estudar 🌸" }, { tempo: 120000, texto: "chega de foto..." }, { tempo: 240000, texto: "já deu de ficar olhando isso 👀" }, { tempo: 360000, texto: "fecha isso e vai fazer alguma coisa útil ♡" }, { tempo: 600000, texto: "sério que você ainda está aqui?" } ]; /* FRASES EXTRAS Usaremos para buscas e outras interações. */ const mensagensExtras = [ "curiosa demais hoje, não?", "essa busca foi muito específica 👀", "tem certeza que quer continuar procurando?", "algumas fotos deveriam continuar esquecidas 😂", "mais uma?", "eu vi essa busca...", "você realmente não desiste ♡", "não vou perguntar o motivo dessa busca.", "tem coisa demais aqui dentro 🌸", "memórias desbloqueadas.", "isso está ficando suspeito 👀", "ok... essa eu deixo passar.", "você sabe que pode fechar o site, né?", "mais fotos. claro.", "sweet little stalker ♡" ]; /* ===================================================== MOSTRAR TOAST ===================================================== */ function mostrarToast(texto) { const container = document.getElementById( "toastContainer" ); if (!container) return; /* Remove toast anterior */ const anterior = container.querySelector( ".toast" ); if (anterior) { anterior.remove(); } const toast = document.createElement( "div" ); toast.className = "toast"; toast.innerHTML = ` ♡ ${texto} `; container.appendChild( toast ); const fechar = toast.querySelector( ".toast-fechar" ); fechar.addEventListener( "click", function() { fecharToast( toast ); } ); /* Pequeno atraso para ativar animação CSS */ requestAnimationFrame( function() { requestAnimationFrame( function() { toast.classList.add( "visivel" ); } ); } ); /* Desaparece após 5 segundos */ setTimeout( function() { fecharToast( toast ); }, 5000 ); } /* ===================================================== FECHAR TOAST ===================================================== */ function fecharToast(toast) { if ( !toast || !toast.isConnected ) return; toast.classList.remove( "visivel" ); toast.classList.add( "saindo" ); setTimeout( function() { if ( toast.isConnected ) { toast.remove(); } }, 350 ); } /* ===================================================== INICIAR RELÓGIO O relógio começa SOMENTE depois que ela acertar o quiz. ===================================================== */ function iniciarMensagens() { /* Impede iniciar duas vezes */ if (inicioNoSite) return; inicioNoSite = Date.now(); mensagensPorTempo.forEach( function(item) { const timer = setTimeout( function() { mostrarToast( item.texto ); }, item.tempo ); timersMensagens.push( timer ); } ); } /* ===================================================== MENSAGEM ALEATÓRIA ===================================================== */ function mensagemAleatoria() { let disponiveis = mensagensExtras.filter( function(frase) { return !mensagensUsadas.has( frase ); } ); /* Se todas já apareceram, reinicia a lista. */ if ( disponiveis.length === 0 ) { mensagensUsadas.clear(); disponiveis = [...mensagensExtras]; } const indice = Math.floor( Math.random() * disponiveis.length ); const frase = disponiveis[indice]; mensagensUsadas.add( frase ); return frase; } /* ===================================================== CONTADOR DE BUSCAS ===================================================== */ function registrarBusca() { contadorBuscas++; /* 3ª busca */ if ( contadorBuscas === 3 ) { mostrarToast( "já é a terceira busca 👀" ); } /* 5ª busca */ else if ( contadorBuscas === 5 ) { mostrarToast( "curiosa demais hoje, não?" ); } /* 8ª busca */ else if ( contadorBuscas === 8 ) { mostrarToast( "o mundo lá fora ainda existe 🌸" ); } /* Depois da 10ª: a cada 4 buscas existe uma mensagem. */ else if ( contadorBuscas >= 10 && contadorBuscas % 4 === 0 ) { mostrarToast( mensagemAleatoria() ); } }
